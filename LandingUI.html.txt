<script>
  let selectedOpponent = {
    id: 'noob',
    name: 'نوب اسب',
    power: 20
  };

  function openSoonModal(featureName) {
    const backdrop = document.getElementById('soonModalBackdrop');
    const titleEl = document.getElementById('soonModalTitle');
    titleEl.textContent = `«${featureName}» به‌زودی اضافه می‌شود!`;
    backdrop.classList.add('active');
  }

  function closeSoonModal() {
    document.getElementById('soonModalBackdrop').classList.remove('active');
  }

  function openBotModal(e) {
    if (e) e.preventDefault();
    document.getElementById('botModalBackdrop').classList.add('active');
    document.body.style.overflow = 'hidden';
  }

  function closeBotModal() {
    document.getElementById('botModalBackdrop').classList.remove('active');
    document.body.style.overflow = '';
  }

  function handleBackdropClick(e) {
    if (e.target.id === 'botModalBackdrop') {
      closeBotModal();
    }
  }

  function selectBot(cardElement) {
    const allCards = document.querySelectorAll('.bot-card');
    allCards.forEach(c => c.classList.remove('selected'));
    
    cardElement.classList.add('selected');
    
    selectedOpponent.id = cardElement.getAttribute('data-bot');
    selectedOpponent.name = cardElement.getAttribute('data-name');
    
    const mainBtn = document.getElementById('mainStartBtn');
    mainBtn.textContent = `ادامه و انتخاب میز نبرد (${selectedOpponent.name}) ←`;
  }

  function goToWorldModal() {
    document.getElementById('botModalBackdrop').classList.remove('active');
    document.getElementById('chosenBotNameLabel').textContent = selectedOpponent.name;
    document.getElementById('worldModalBackdrop').classList.add('active');
  }

  function closeWorldModal() {
    document.getElementById('worldModalBackdrop').classList.remove('active');
    document.body.style.overflow = '';
  }

  function backToBotModal() {
    document.getElementById('worldModalBackdrop').classList.remove('active');
    document.getElementById('botModalBackdrop').classList.add('active');
  }

  function enterGameArena(worldId) {
    const gameConfig = {
      bot: selectedOpponent,
      world: worldId
    };
    sessionStorage.setItem('selectedChessBot', JSON.stringify(selectedOpponent));
    sessionStorage.setItem('shahmatGameConfig', JSON.stringify(gameConfig));

    closeWorldModal();
    window.location.href = `ChessGame.html?bot=${selectedOpponent.id}&world=${worldId}`;
  }

  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
      closeSoonModal();
      closeBotModal();
      closeWorldModal();
    }
  });
</script>
